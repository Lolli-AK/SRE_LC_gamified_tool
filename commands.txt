**create room and join twice:**

ROOM_ID=$(curl -s -X POST http://localhost:8000/rooms | python -c "import sys,json; print(json.load(sys.stdin)['room_id'])")
P1=$(curl -s -X POST http://localhost:8000/rooms/$ROOM_ID/join -H "Content-Type: application/json" -d '{"name":"P1"}' | python -c "import sys,json; print(json.load(sys.stdin)['player_id'])")
P2=$(curl -s -X POST http://localhost:8000/rooms/$ROOM_ID/join -H "Content-Type: application/json" -d '{"name":"P2"}' | python -c "import sys,json; print(json.load(sys.stdin)['player_id'])")
echo "ROOM=$ROOM_ID"
echo "P1=$P1"
echo "P2=$P2"

wscat -c "ws://localhost:8000/ws/$ROOM_ID?player_id=$P1"
